cmake_minimum_required(VERSION 3.16)
project(observer_workshop LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(observer_demo main.cpp)
add_executable(observer_check check.cpp)
if(MSVC)
  target_compile_options(observer_demo PRIVATE /W4 /utf-8)
  target_compile_options(observer_check PRIVATE /W4 /utf-8)
else()
  target_compile_options(observer_demo PRIVATE -Wall -Wextra)
  target_compile_options(observer_check PRIVATE -Wall -Wextra)
endif()
enable_testing()
add_test(NAME observer_contract COMMAND observer_check)
